المشكلة: الروابط لا تعمل لأن Docker Desktop غير مشغّل

الحل السريع:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1️⃣  افتح Docker Desktop
   - ابحث عن "Docker Desktop" في قائمة البرامج
   - انقر مرتين لفتح التطبيق

2️⃣  انتظر حتى يصبح جاهزاً
   - ستظهر أيقونة Docker في شريط النظام
   - انتظر حتى تتحول للون الأخضر
   - قد يستغرق 30-60 ثانية

3️⃣  تحقق من التشغيل
   افتح PowerShell واكتب:
   
   docker ps
   
   يجب أن ترى جدول بدون أخطاء

4️⃣  شغّل السكريبت مرة أخرى
   
   .\setup_local.ps1

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

بعد تشغيل Docker واكتمال البناء، ستعمل الروابط التالية:

✅ http://localhost:8080      ← التطبيق الرئيسي
✅ http://localhost:4400      ← Firebase Admin UI
✅ http://localhost:8081      ← Firestore Database UI
✅ http://localhost:9099      ← Auth Emulator
✅ http://localhost:5001      ← Cloud Functions

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

لماذا الروابط لا تعمل الآن؟
• لأن الحاويات (Containers) غير مشغّلة
• الحاويات تحتاج Docker Desktop ليعمل
• بمجرد تشغيل Docker وبناء المشروع، ستعمل الروابط تلقائياً
